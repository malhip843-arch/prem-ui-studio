  <div class="signup">
    <h2>Create Account</h2>

    <form id="signupForm">
      <input id="name" placeholder="Full Name" required>
      <input id="email" type="email" placeholder="Email Address" required>
      <input id="phone" type="tel" placeholder="Phone (+923001234567)" required>
      <input id="password" type="password" placeholder="Password (6+ characters)" required minlength="6">
      <input id="confirmPassword" type="password" placeholder="Confirm Password" required>
      <button type="submit" id="sendOtp">Create Account & Send OTP</button>
    </form>

    <div id="recaptcha-container"></div>

    <div id="otpSection">
      <p>Enter the 6-digit OTP sent to your phone.</p>
      <input id="otp" type="text" inputmode="numeric" maxlength="6" placeholder="Enter OTP">
      <button id="verifyOtp" type="button">Verify OTP</button>
    </div>

    <p id="message"></p>
  </div>

  <script type="module">
    import { auth } from "./firebase.js";

    import {
      createUserWithEmailAndPassword,
      updateProfile,
      RecaptchaVerifier,
      linkWithPhoneNumber
    } from "https://www.gstatic.com/firebasejs/13.4.0/firebase-auth.js";

    const form = document.getElementById("signupForm");
    const message = document.getElementById("message");
    const otpSection = document.getElementById("otpSection");

    let confirmationResult = null;

    form.addEventListener("submit", async (e) => {
      e.preventDefault();

      const name = document.getElementById("name").value.trim();
      const email = document.getElementById("email").value.trim();
      const phone = document.getElementById("phone").value.trim();
      const password = document.getElementById("password").value;
      const confirmPassword = document.getElementById("confirmPassword").value;

      if (password !== confirmPassword) {
        message.textContent = "Passwords do not match.";
        return;
      }

      if (!/^\+[1-9]\d{7,14}$/.test(phone)) {
        message.textContent = "Phone must include country code, e.g. +923001234567";
        return;
      }

      try {
        message.textContent = "Creating your account...";

        const result = await createUserWithEmailAndPassword(
          auth, email, password
        );

        await updateProfile(result.user, {
          displayName: name
        });

        const verifier = new RecaptchaVerifier(
          auth,
          "recaptcha-container",
          { size: "normal" }
        );

        confirmationResult = await linkWithPhoneNumber(
          result.user,
          phone,
          verifier
        );

        otpSection.style.display = "block";
        form.style.display = "none";
        message.textContent = "OTP sent! Please verify your phone.";

      } catch (error) {
        message.textContent = error.message;
      }
    });

    document.getElementById("verifyOtp").addEventListener("click", async () => {
      const code = document.getElementById("otp").value.trim();

      if (!confirmationResult || !/^\d{6}$/.test(code)) {
        message.textContent = "Please enter the 6-digit OTP.";
        return;
      }

      try {
        await confirmationResult.confirm(code);
        message.textContent = "Account created and phone verified successfully!";
        otpSection.style.display = "none";
      } catch (error) {
        message.textContent = "OTP verification failed: " + error.message;
      }
    });
  </script>
